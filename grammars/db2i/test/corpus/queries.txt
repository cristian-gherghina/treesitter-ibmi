==================
Recursive common table expression
==================

with recursive hierarchy(id, parent_id, depth) as (
  select id, parent_id, 1 from nodes where parent_id is null
  union all
  select n.id, n.parent_id, h.depth + 1
    from nodes n join hierarchy h on n.parent_id = h.id
)
select * from hierarchy;

==================
Window function and CASE
==================

select customer,
       case when balance < 0 then 'DUE' else 'OK' end as state,
       row_number() over (partition by region order by balance desc) as position
  from accounts;

  from sales
 group by grouping sets ((region, department), (region), ());

==================
Correlated subquery
==================

select c.customer_id
  from customers c
 where exists (select 1 from orders o where o.customer_id = c.customer_id);
